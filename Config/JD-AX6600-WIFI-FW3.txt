# FW3 覆盖文件 — 仅包含与 FW4 基线不同的行
# 使用方法：workflow 中选择 fw3 时自动追加到 .config

# --- 防火墙栈切换（FW3 iptables）---
CONFIG_PACKAGE_firewall4=n                  # 关闭 FW4
CONFIG_PACKAGE_firewall=y                   # 启用 FW3
CONFIG_PACKAGE_iptables=y                   # iptables 用户态
CONFIG_PACKAGE_ip6tables=y                  # IPv6 iptables
CONFIG_PACKAGE_ip6tables-extra=y            # IPv6 扩展模块
CONFIG_PACKAGE_ip6tables-mod-nat=y          # IPv6 NAT 模块
CONFIG_PACKAGE_kmod-ipt-fullconenat=y       # FullCone NAT 内核模块
CONFIG_PACKAGE_iptables-mod-fullconenat=y   # FullCone NAT 用户态模块

# --- 代理插件状态切换 ---
CONFIG_PACKAGE_luci-app-homeproxy=n         # FW3 下不启用 HomeProxy
CONFIG_PACKAGE_luci-app-turboacc=y          # FW3 默认启用 Turbo ACC
CONFIG_PACKAGE_luci-i18n-turboacc-zh-cn=y   # Turbo ACC 中文语言包
